  name: momentumBonusHit,
  parameters: [movedSectors : Nat],

definition{
  name: resolveOpposedNetHits,
  parameters: [rawAtkHits : Nat, momentumSectors : Nat, rawDefHits : Nat, coverLevel : Nat],
  result: Nat,
  body: (if (2 <=? coverLevel) then 0 else (prim subtract (rawAtkHits + (momentumBonusHit momentumSectors)) (rawDefHits + (coverAutoEvade coverLevel)) : Nat)),
  axioms: []
}

definition{
  name: checkStrikeFixture,
  parameters: [f : StrikeFixture],
  result: Bool,
  body: (((resolveOpposedNetHits (project f rawAtkHits) (project f momentumSectors) (project f rawDefHits) (project f coverLevel)) =? (project f expectedNetHits)) && ((resolveMitigatedTrauma (project f expectedNetHits) (project f targetPlating) (project f targetResilience)) =? (project f expectedFinalTrauma))),
  axioms: []
}

definition{
  name: allStrikeFixturesPass,
  parameters: [xs : (List StrikeFixture)],
  result: Bool,
  body: (match xs
    | List.nil => true
    | List.cons head tail => (if (checkStrikeFixture head) then (allStrikeFixturesPass tail) else false)),
  axioms: [],
  recursive_argument: xs
}
