definition{
  name: sameDigest,
  parameters: [left : Digest, right : Digest],
  result: Bool,
  body: (((project left w0) =? (project right w0)) && (((project left w1) =? (project right w1)) && (((project left w2) =? (project right w2)) && ((project left w3) =? (project right w3)))))
}

definition{
  name: SameDigest,
  parameters: [left : Digest, right : Digest],
  result: Prop,
  body: (((project left w0) = (project right w0)) /\ (((project left w1) = (project right w1)) /\ (((project left w2) = (project right w2)) /\ ((project left w3) = (project right w3)))))
}

definition{
  name: pinMatches,
  parameters: [pin : Pin],
  result: Bool,
  body: (sameDigest (project pin expected) (project pin observed))
}

definition{
  name: conformant,
  parameters: [observation : Observation],
  result: Bool,
  body: ((pinMatches (project observation devcontainer)) && ((pinMatches (project observation bootstrap)) && ((pinMatches (project observation agents)) && ((pinMatches (project observation verification)) && ((pinMatches (project observation prismpmLock)) && (pinMatches (project observation templateContract)))))))
}

definition{
  name: Conformant,
  parameters: [observation : Observation],
  result: Prop,
  body: ((SameDigest (project (project observation devcontainer) expected) (project (project observation devcontainer) observed)) /\ ((SameDigest (project (project observation bootstrap) expected) (project (project observation bootstrap) observed)) /\ ((SameDigest (project (project observation agents) expected) (project (project observation agents) observed)) /\ ((SameDigest (project (project observation verification) expected) (project (project observation verification) observed)) /\ ((SameDigest (project (project observation prismpmLock) expected) (project (project observation prismpmLock) observed)) /\ (SameDigest (project (project observation templateContract) expected) (project (project observation templateContract) observed)))))))
}

theorem{
  name: conformant_decides,
  parameters: [observation : Observation],
  statement: (((conformant observation) = true) <-> (Conformant observation)),
  proof: simplify{
    definitions: [
      Bool.and_eq_true,
      Conformant,
      Nat.beq_eq,
      SameDigest,
      conformant,
      pinMatches,
      sameDigest
    ]
  },
  axioms: [propext]
}

definition{
  name: digestFrom,
  parameters: [value : Bytes, offset : Nat, remaining : Nat, accumulated : Nat],
  result: Nat,
  body: (match remaining
    | Nat.zero => accumulated
    | Nat.succ rest => (match (Foundation.Bytes::byteAt value offset)
      | Option.none => accumulated
      | Option.some byte => (digestFrom value (offset + 1) rest ((prim multiply accumulated 256 : Nat) + (prim to_nat byte : Nat))))),
  axioms: [Quot.sound, propext],
  recursive_argument: remaining
}

definition{
  name: digestAt,
  parameters: [value : Bytes, offset : Nat],
  result: Digest,
  body: (record Digest {w0 := (digestFrom value offset 8 0), w1 := (digestFrom value (offset + 8) 8 0), w2 := (digestFrom value (offset + 16) 8 0), w3 := (digestFrom value (offset + 24) 8 0)}),
  axioms: [Quot.sound, propext]
}

  name: pinAt,

  result: Pin,

definition{
  name: driftEntry,
  parameters: [pin : Pin, path : Bytes],
  result: Bytes,
  body: (if (pinMatches pin) then 0x else path)

}

definition{
  name: dispatch,
  parameters: [value : Bytes],
  result: Bytes,
  body: (if ((Foundation.Bytes::byteLength value) =? 384) then (if (conformant (observe value)) then (prim utf8_encode "conformant" : Bytes) else (prim append (prim utf8_encode "drift:" : Bytes) (drifted (observe value)) : Bytes)) else (prim utf8_encode "invalid request" : Bytes)),
  axioms: [Quot.sound, propext]
}
